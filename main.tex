\section{Problem Formulation and Method}
\label{sec:method}
Consider the observation model
\begin{equation}
  \vect{y}_k = \mat{H}\,\vect{x}_k + \vect{n}_k,
  \qquad \vect{n}_k \sim \mathcal{N}(\vect{0}, \mat{R}),
  \label{eq:model}
\end{equation}
where $\vect{x}_k \in \R^n$ is the state. The estimate is obtained by
\begin{equation}
  \hat{\vect{x}} = \argmin_{\vect{x}} \; \lVert \vect{y} - \mat{H}\vect{x} \rVert_{\mat{R}^{-1}}^2 .
\end{equation}

\begin{algorithm}[htbp]
  \caption{Proposed procedure}
  \label{alg:main}
  \begin{algorithmic}[1]
    \Require Measurements $\{\vect{y}_k\}_{k=1}^{K}$
    \Ensure Estimate $\hat{\vect{x}}$
    \State Initialize $\hat{\vect{x}}_0$
    \For{$k = 1$ to $K$}
      \State Update estimate using \cref{eq:model}
    \EndFor
    \State \Return $\hat{\vect{x}}_K$
  \end{algorithmic}
\end{algorithm}
